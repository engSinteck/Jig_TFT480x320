from PIL import Image, ImageOps

def redimensionar_para_canvas(caminho_imagem, caminho_saida, largura_alvo=179, altura_alvo=36):
    img = Image.open(caminho_imagem).convert("RGBA")
    
    # Redimensiona mantendo a proporção para caber dentro de 179x36
    img.thumbnail((largura_alvo, altura_alvo), Image.Resampling.LANCZOS)
    
    # Cria uma nova imagem com o tamanho exato de 179x36 com fundo transparente
    canvas = Image.new("RGBA", (largura_alvo, altura_alvo), (0, 0, 0, 0))
    
    # Centraliza o botão dentro do canvas 179x36
    pos_x = (largura_alvo - img.width) // 2
    pos_y = (altura_alvo - img.height) // 2
    canvas.paste(img, (pos_x, pos_y), img)
    
    # Salva como PNG transparente
    canvas.save(caminho_saida, "PNG")

# Execute para cada uma das suas 3 imagens originais:
redimensionar_para_canvas("botao_normal.png", "btn_normal_179x36.png")
redimensionar_para_canvas("botao_apertado.png", "btn_pressed_179x36.png")
redimensionar_para_canvas("botao_selecionado.png", "btn_selected_179x36.png")
